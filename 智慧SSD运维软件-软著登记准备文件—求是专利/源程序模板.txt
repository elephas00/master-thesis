<script setup lang="ts">
import { ref, reactive } from 'vue'

const emit = defineEmits<{
  login: [username: string, password: string]
  register: [username: string, password: string]
}>()

const isLoginMode = ref(true)
const formData = reactive({
  username: '',
  password: ''
})
const errorMessage = ref('')

function toggleMode() {
  isLoginMode.value = !isLoginMode.value
  errorMessage.value = ''
  formData.username = ''
  formData.password = ''
}

function handleSubmit() {
  if (!formData.username.trim()) {
    errorMessage.value = '请输入用户名'
    return
  }
  if (!formData.password.trim()) {
    errorMessage.value = '请输入密码'
    return
  }
  
  errorMessage.value = ''
  
  if (isLoginMode.value) {
    emit('login', formData.username, formData.password)
  } else {
    emit('register', formData.username, formData.password)
  }
}

function generateId() {
  return Math.random().toString(36).slice(2) + Date.now().toString(36);

export async function uploadDisks(rows: any[]): Promise<{ inserted: number }>{
  const normalized: Disk[] = rows.map((r) => ({
    id: generateId(),
    serial: String(r.serial ?? r.Serial ?? r.SN ?? ''),
    model: String(r.model ?? r.Model ?? ''),
    capacityBytes: Number(r.capacityBytes ?? r.CapacityBytes ?? r.capacity ?? 0),
    vendor: r.vendor ?? r.Vendor,
    smart: r.smart ?? r.SMART ?? r.smartData ?? undefined,
    createdAt: new Date().toISOString(),
  })).filter(d => d.serial);
  disks = normalized.concat(disks);
  return { inserted: normalized.length };
}

export function computeFaultScore(d: Disk): { score: number; reasons: string[] } {
  const s = (k: string) => Number((d.smart?.[k] as any) ?? 0) || 0;
  const reasons: string[] = [];
  if (s('Reallocated_Sector_Ct') > 0) reasons.push('Reallocated_Sector_Ct>0');
  if (s('Current_Pending_Sector') > 0) reasons.push('Current_Pending_Sector>0');
  if (s('Offline_Uncorrectable') > 0) reasons.push('Offline_Uncorrectable>0');
  if (s('Temperature_Celsius') > 55) reasons.push('Temp>55');
  const poh = s('Power_On_Hours');
  if (poh > 40000 && (s('Reallocated_Sector_Ct')>0 || s('Current_Pending_Sector')>0)) reasons.push('High POH + issues');
  return { score: reasons.length, reasons };
}

function sortData<T extends Record<string, any>>(arr: T[], sortBy?: string, order?: 'asc'|'desc') {
  if (!sortBy) return arr;
  const mult = order === 'desc' ? -1 : 1;
  return [...arr].sort((a,b) => {
    const av = a[sortBy]; const bv = b[sortBy];
    if (av === bv) return 0;
    return av > bv ? mult : -mult;
  });
}

export async function listDisks(p: PageParams): Promise<PageResult<Disk>> {
  const { page, pageSize, q, sortBy, order } = p;
  const filtered = (q? disks.filter(d =>
    d.serial.includes(q) || d.model.includes(q) || (d.vendor??'').includes(q)
  ): disks);
  const sorted = sortData(filtered, sortBy, order);
  const start = (page-1)*pageSize;
  const data = sorted.slice(start, start+pageSize);
  return { data, total: filtered.length };
}

export async function listFaults(p: PageParams): Promise<PageResult<Disk & { score: number; reasons: string[] }>> {
  const withScores = disks.map(d => ({ ...d, ...computeFaultScore(d) }));
  const faults = withScores.filter(d => d.score >= 2);
  const { page, pageSize, q, sortBy, order } = p;
  const filtered = q? faults.filter(d => d.serial.includes(q) || d.model.includes(q)) : faults;
  const sorted = sortData(filtered as any[], sortBy, order) as any[];
  const start = (page-1)*pageSize;
  const data = sorted.slice(start, start+pageSize) as any[];
  return { data, total: filtered.length };
}
